package photo

// T-056 review round 1, M2 (a scratch probe, not committed): the process's
// peak while 6 uploads of the worst accepted kind are sanitized at once.

import (
	"context"
	"os"
	"path/filepath"
	"runtime"
	"sync"
	"sync/atomic"
	"testing"
	"time"
)

func TestR1Concurrent(t *testing.T) {
	dir := os.Getenv("R1_IMGS")
	if dir == "" {
		t.Skip("R1_IMGS not set")
	}
	name := os.Getenv("R1_FILE")
	if name == "" {
		name = "rgba8.png"
	}
	data, err := os.ReadFile(filepath.Join(dir, name))
	if err != nil {
		t.Fatal(err)
	}
	runtime.GC()
	var base runtime.MemStats
	runtime.ReadMemStats(&base)
	var peak atomic.Uint64
	done := make(chan struct{})
	stopped := make(chan struct{})
	go func() {
		defer close(stopped)
		var m runtime.MemStats
		for {
			runtime.ReadMemStats(&m)
			if m.HeapInuse > peak.Load() {
				peak.Store(m.HeapInuse)
			}
			select {
			case <-done:
				return
			case <-time.After(time.Millisecond):
			}
		}
	}()
	var wg sync.WaitGroup
	start := time.Now()
	for range 6 {
		wg.Go(func() {
			if _, err := Sanitize(context.Background(), data); err != nil {
				t.Error(err)
			}
		})
	}
	wg.Wait()
	close(done)
	<-stopped
	t.Logf("%s: 6 at once (%d slots): %.0f ms, peak heap in use +%d MiB", name, cap(decodeSlots), float64(time.Since(start).Milliseconds()), (peak.Load()-base.HeapInuse)>>20)
}
