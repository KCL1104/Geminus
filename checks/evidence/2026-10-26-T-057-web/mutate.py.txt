def main() -> int:
    group, out = sys.argv[1], Path(sys.argv[2])
    rows = []
    for name, rel, old, new, tests in MUTANTS[group]:
        path = WEB / rel
        original = path.read_text(encoding="utf-8")
        if original.count(old) != 1:
            rows.append((name, "INVALID (pattern not found once)"))
            continue
        try:
            path.write_text(original.replace(old, new), encoding="utf-8")
            r = subprocess.run(["npx", "vitest", "run", *tests], cwd=WEB, capture_output=True, text=True, timeout=900)
            rows.append((name, "killed" if r.returncode != 0 else "SURVIVED"))
        finally:
            path.write_text(original, encoding="utf-8")
        print(rows[-1], flush=True)
    out.write_text("".join(f"{n}\t{s}\n" for n, s in rows), encoding="utf-8")
    return 0


if __name__ == "__main__":
    sys.exit(main())
