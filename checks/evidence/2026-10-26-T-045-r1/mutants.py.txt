def run(web: pathlib.Path) -> tuple[int, str]:
    p = subprocess.run(["npx", "vitest", "run"], cwd=web, capture_output=True, text=True, timeout=900)
    out = re.sub(r"\x1b\[[0-9;]*m", "", p.stdout + p.stderr)
    summary = [l.strip() for l in out.splitlines() if re.match(r"\s*(Test Files|Tests)\s", l)]
    fails = sorted({l.strip() for l in out.splitlines() if l.strip().startswith("FAIL ")})
    return p.returncode, "\n".join(["   " + s for s in summary] + ["   " + f for f in fails[:12]] + ([f"   … {len(fails) - 12} more"] if len(fails) > 12 else []))


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("web")
    ap.add_argument("--tests", choices=["head", "new"], required=True)
    ap.add_argument("--head-tests")
    ap.add_argument("ids", nargs="*")
    a = ap.parse_args()
    web = pathlib.Path(a.web).resolve()
    swapped: dict[pathlib.Path, bytes] = {}
    if a.tests == "head":
        for name in ("simplicity.test.tsx", "tokens.test.ts"):
            target = web / "tests" / name
            swapped[target] = target.read_bytes()
            target.write_bytes((pathlib.Path(a.head_tests) / name).read_bytes())
    worst = 0
    try:
        for mid in a.ids or list(MUTANTS):
            originals: dict[pathlib.Path, bytes] = {}
            try:
                for rel, old, new in MUTANTS[mid]:
                    f = (web / rel).resolve()
                    originals.setdefault(f, f.read_bytes())
                    text = f.read_text()
                    if text.count(old) != 1:
                        raise SystemExit(f"{mid}: pattern not found exactly once in {rel}")
                    f.write_text(text.replace(old, new))
                code, report = run(web)
            finally:
                for f, b in originals.items():
                    f.write_bytes(b)
            verdict = "KILLED" if code != 0 else "SURVIVED"
            if code == 0:
                worst = 1
            print(f"{mid} (tests: {a.tests}) -> vitest exit {code}: {verdict}\n{report}\n", flush=True)
    finally:
        for f, b in swapped.items():
            f.write_bytes(b)
    return worst


if __name__ == "__main__":
    sys.exit(main())
