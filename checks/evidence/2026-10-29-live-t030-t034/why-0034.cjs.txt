async function until(page, scope, ends, ms) {
  for (const deadline = Date.now() + ms; Date.now() < deadline; await page.waitForTimeout(3000)) {
    const text = await scope.innerText().catch(() => "");
    const hit = ends.find((e) => text.includes(e));
    if (hit) return hit;
  }
  return "timeout";
}
