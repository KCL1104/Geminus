// P6. D-097 (f): setup_ticks is the integral of Z over [W, H]; a changeover
// ends with no observation of its own (SIMULATION_SPEC §5: the collector
// keeps the ends in a min-heap). G-SU1 over [0, 49]: changeovers 20-23,
// 25-28, 40-43 and 45-48 (by hand from G-SU1's derivation), so setup_ticks
// 12 min = 720000 ticks, 4 started, 4 complete. A4's changeover ends at 48
// and its service at 50 > H, so no observation comes after 48: only
// Collector.Finish applies that end. The reviewer's mutant
// r-setup-ends-not-applied-at-finish leaves Z at 1 over [48, 49].
func TestProbeChangeoverEndingAfterTheLastObservation(t *testing.T) {
	var doc struct {
		Cases []goldenCase `json:"cases"`
	}
	if err := json.Unmarshal(contracttest.ReadFile(t, filepath.Join(contracttest.Root(t), "testdata/simulation/setup-goldens.json")), &doc); err != nil {
		t.Fatal(err)
	}
	var c goldenCase
	for _, x := range doc.Cases {
		if x.ID == "G-SU1" {
			c = x
		}
	}
	in := caseInput(t, c)
	in.Config.Warmup.Value, in.Config.Warmup.Unit = 0, model.Minute
	in.Config.Horizon.Value, in.Config.Horizon.Unit = 49, model.Minute
	res := run(t, &Engine{}, in).Result
	for id, st := range res.ReplicationSummaries[0].ResourceStates {
		t.Logf("%s: setup_ticks %d setups_started %d setups_completed %d completed_setup_ticks %d scheduled_processing_ticks %d in_use_ticks %d",
			id, *st.SetupTicks, *st.SetupsStarted, *st.SetupsCompleted, *st.CompletedSetupTicks, st.ScheduledProcessingTicks, st.InUseTicks)
		if *st.SetupTicks != 720000 || *st.SetupsStarted != 4 || *st.SetupsCompleted != 4 {
			t.Errorf("%s: by hand setup_ticks 720000 (12 min), 4 started, 4 complete", id)
		}
	}
}
